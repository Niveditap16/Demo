import com.fasterxml.jackson.databind.ObjectMapper;
import com.mks.api.CmdRunner;
import com.mks.api.Command;
import com.mks.api.IntegrationPoint;
import com.mks.api.IntegrationPointFactory;
import com.mks.api.Option;

public class App {

    public CmdRunner cmdRunner;

    public App(Session session) throws APIException {
        this.cmdRunner = session.createCmdRunner();
    }

    public CmdRunner getCmdRunner() {
        return this.cmdRunner;
    }

    public static void main(String[] args) throws APIException {

        Properties props = new Properties();
        try (FileInputStream fis = new FileInputStream("mks.properties")) {
            props.load(fis);

            IntegrationPointFactory ipf = IntegrationPointFactory.getInstance();
            IntegrationPoint integrationPoint = ipf.createLocalIntegrationPoint((VersionNumber) APIVersion.API_4_16);
            integrationPoint.setAutoStartIntegrityClient(true);
   Session session = integrationPoint.getCommonSession();
   session.setDefaultHostname(props.getProperty("hostname"));
   session.setDefaultPort(Integer.parseInt(props.getProperty("port")));
   session.setDefaultUsername(props.getProperty("username"));

            App app = new App(session);

            List<WorkItem> documentTypes = app.getAllDocumentTypes();
            Map<String, Set<String>> typettributes = app.getAllAttributes(documentTypes);
            Set<String> attributenames = new HashSet<>();
            typettributes.keySet().forEach(key -> attributenames.addAll(typettributes.get(key)));

            Map<String, Map<String, Object>> data = app.getAttributeMetadata(attributenames);

            File file = new File("integrity.json");

            Map<String, List<Map<String, Object>>> datamodel = new HashMap<>();
            Iterator<String> itr = typettributes.keySet().iterator();
            while (itr.hasNext()) {
                String type = itr.next();
                Set<String> attmap = typettributes.get(type);

                List<Map<String, Object>> typeattlist = new ArrayList<>();

                Iterator<String> attitr = attmap.iterator();
                while (attitr.hasNext()) {
                    String att = attitr.next();
                    typeattlist.add(data.get(att));
                }
                datamodel.put(type, typeattlist);
            }

            ObjectMapper om = new ObjectMapper();
            om.writeValue(file, datamodel);

            session.release();

        } catch (IOException e) {
            e.printStackTrace();
        } catch (APIException e) {
            System.err.println("unable to initiate connection");
            System.err.println(e.getMessage());
        }

    }

    private Map<String, Map<String, Object>> getAttributeMetadata(Set<String> attributenames) throws APIException {
        Map<String, Map<String, Object>> metadata = new HashMap<>();
        for (String attributename : attributenames) {
            Response response = viewfield(attributename);
            WorkItemIterator wii = response.getWorkItems();
            while (wii.hasNext()) {
                WorkItem wi = wii.next();

                Field type = wi.getField("type");
                Field computation = null;
                if (wi.contains("computation")) {
                    computation = wi.getField("computation");
                }
                Field richcontent = null;
                if (wi.contains("richContent")) {
                    richcontent = wi.getField("richContent");
                }

                String fname, ftype;
                List<String> fpicks = null;
                String fbackedby = null;
                boolean isrich, iscompuation;
                boolean ismultivalued = false;

                fname = wi.getId();
                ftype = type.getValueAsString();
                if (richcontent != null) {
                    isrich = richcontent.getBoolean();
                } else {
                    isrich = false;
                }
                if (computation != null && computation.getValue() != null) {
                    iscompuation = true;
                } else {
                    iscompuation = false;
                }
                switch (type.getValueAsString()) {
                    case "pick":
                        fpicks = Arrays.asList(wi.getField("picks").getValueAsString().split(","));
                        ismultivalued = wi.getField("isMultiValued").getBoolean();
                        break;
                    case "ibpl":
                        fpicks = Arrays.asList(wi.getField("picks").getValueAsString().split(","));
                        ismultivalued = wi.getField("isMultiValued").getBoolean();
                        break;
                    case "fva":
                        String backedbyatt = wi.getField("backedBy").getValueAsString().split("\\.")[1];
                        fbackedby = backedbyatt;
                        Set<String> set = new HashSet<>();
                        set.add(backedbyatt);
                        Map<String, Map<String, Object>> map = getAttributeMetadata(set);
                        Map<String, Object> attmetadata = map.get(backedbyatt);
                        switch ((String) attmetadata.get("type")) {
                        case "pick":
                         fpicks = (List<String>) attmetadata.get("fpicks");
                            ismultivalued = (boolean) attmetadata.get("ismultivalued");
                            ftype = (String) attmetadata.get("type");
                            break;
                        case "ibpl":
                         fpicks = (List<String>) attmetadata.get("fpicks");
                            ismultivalued = (boolean) attmetadata.get("ismultivalued");
                            ftype = (String) attmetadata.get("type");
                            break;
                        default:
                         ftype = (String) attmetadata.get("type");
                         break;
                        }
                        break;
                    default:
                        break;
                }
                System.out.println(
                        fname + " " + ftype + " " + iscompuation + " " + isrich + " " + fpicks + " " + ismultivalued + " " + fbackedby);
                Map<String, Object> row = new HashMap<>();
                row.put("name", fname);
                row.put("type", ftype);
                row.put("iscomputation", iscompuation);
                row.put("isrich", isrich);
                row.put("fpicks", fpicks);
                row.put("ismultivalued", ismultivalued);
                metadata.put(attributename, row);
            }
        }
        return metadata;
    }

    private Response viewfield(String attributename) {
        Command fields = new Command(Command.IM, "viewfield");
        fields.addOption(new Option("hostname",
                "integrity-rail.corp.knorr-bremse.com"));
        fields.addSelection(attributename);
        Response response = runCommand(fields);
        return response;
    }

    private Map<String, Set<String>> getAllAttributes(List<WorkItem> documentTypes) throws APIException {
        Map<String, Set<String>> typeattributemap = new HashMap<>();

        for (WorkItem type : documentTypes) {
            Set<String> attributenames = new HashSet<>();
            Command typeatts = new Command(Command.IM, "viewtype");
            typeatts.addOption(new Option("hostname",
                    "integrity-rail.corp.knorr-bremse.com"));
            typeatts.addOption(new Option("port", Integer.toString(7001)));
            typeatts.addSelection(type.getField("name").getValueAsString());
            Response response2 = runCommand(typeatts);

            WorkItemIterator wii = response2.getWorkItems();
            while (wii.hasNext()) {
                WorkItem wi = wii.next();
                Field visible = wi.getField("visibleFields");
                Field system = wi.getField("systemManagedFields");
                ItemList attlist = (ItemList) visible.getValue();
                Iterator<Item> itr = attlist.getItems();
                while (itr.hasNext()) {
                    Item item = itr.next();
                    attributenames.add(item.getId());
                }
                ItemList attlist2 = (ItemList) system.getValue();
                Iterator<Item> itr2 = attlist2.getItems();
                while (itr2.hasNext()) {
                    Item item = itr2.next();
                    attributenames.add(item.getId());
                }
            }
            typeattributemap.put(type.getId(), attributenames);
        }
        return typeattributemap;
    }

    private List<WorkItem> getAllDocumentTypes() throws APIException {
        Command types = new Command(Command.IM, "types");
        types.addOption(new Option("hostname",
                "integrity-rail.corp.knorr-bremse.com"));
        types.addOption(new Option("port", Integer.toString(7001)));
        types.addOption(new Option("fields", "name,documentClass"));
        types.addOption(new Option("fieldsDelim", ","));
        types.addOption(new Option("batch"));
        List<WorkItem> documentTypes = new ArrayList<>();
        Response response = runCommand(types);
        WorkItemIterator wii = response.getWorkItems();
        while (wii.hasNext()) {
            WorkItem wi = wii.next();
            if (wi.getField("documentClass").getValue().equals("segment")
                    || wi.getField("documentClass").getValue().equals("node"))
                documentTypes.add(wi);
        }
        return documentTypes;
    }

    public synchronized Response runCommand(Command cmd) {
        Response resp = null;
        try {
            resp = this.cmdRunner.execute(cmd);
            return resp;
        } catch (APIException e) {
            System.err.println("Errors while executing command");
            System.err.println(e.getMessage());
        }
        return resp;
    }
}

import com.jamasoftware.services.restclient.jamadomain.lazyresources.PickList;
import com.jamasoftware.services.restclient.jamadomain.lazyresources.PickListOption;

public class Jama {
    public static void main(String[] args) {
        try {
            JamaConfig jamaConfig = new JamaConfig(true);
            JamaInstance jamaInstance = new JamaInstance(jamaConfig);
            List<JamaItemType> types = jamaInstance.getItemTypes();
            Map<String, Set<Map<String, Object>>> datamodel = new HashMap<>();
            for (JamaItemType type : types) {
                Set<Map<String, Object>> fieldset = new HashSet<>();
                
                System.out.println(type.getDisplay());
                List<JamaField> fields = type.getFields();
                for (JamaField field : fields) {
                    Map<String, Object> fielddata = getFieldData(field);
                    fieldset.add(fielddata);
                }

                Map<String, Object> typeid = new HashMap<>();
                typeid.put("id", type.getId());
                fieldset.add(typeid);

                datamodel.put(type.getDisplay(), fieldset);
            }
            wirtetoJson(datamodel);
        } catch (RestClientException e) {
            e.printStackTrace();
        }
    }

 private static void wirtetoJson(Map<String, Set<Map<String, Object>>> datamodel) {
  ObjectMapper om = new ObjectMapper();
        File file = new File("jama.json");
        try {
            om.writeValue(file, datamodel);
        } catch (IOException e) {
            e.printStackTrace();
        }
 }
    
    private static Map<String, Object> getFieldData(JamaField field) throws RestClientException {
     Map<String, Object> fielddata = new HashMap<>();
     fielddata.put("Name", field.getLabel());
     fielddata.put("Type", field.type);
     switch (field.getClass().getName()) {
        case "com.jamasoftware.services.restclient.jamadomain.fields.RichTextField":
            fielddata.put("iscomputation", false);
            fielddata.put("isrich", true);  
            fielddata.put("fpicks", null);
            fielddata.put("ismultivalued", false);
            break;
        case "com.jamasoftware.services.restclient.jamadomain.fields.PickListField":
            fielddata.put("iscomputation", false);
            fielddata.put("isrich", false);    
            fielddata.put("fpicks", getpicklistvaluelist(field));
            fielddata.put("ismultivalued", false);
            break;
        case "com.jamasoftware.services.restclient.jamadomain.fields.MultiSelectField":
            fielddata.put("iscomputation", false);
            fielddata.put("isrich", false);    
            fielddata.put("fpicks", getpicklistvaluelist(field));
            fielddata.put("ismultivalued", true);
            break;
        default:
            fielddata.put("iscomputation", false);
            fielddata.put("isrich", false);    
            fielddata.put("fpicks", null);
            fielddata.put("ismultivalued", false);
            break;
     }
     return fielddata;
 }
    
    private static List<String> getpicklistvaluelist(JamaField field) throws RestClientException {
        List<String> picklist = new ArrayList<>();
        PickList list = null;
        if (field.getClass().getName().equals("com.jamasoftware.services.restclient.jamadomain.fields.PickListField")) {
            list = ((PickListField) field).getPickList();
        } else {
            list = ((MultiSelectField) field).getPickList();
        }

        List<PickListOption> options = list.getOptions();
        options.forEach(o -> picklist.add(o.getName()));
        return picklist;
    }
}
